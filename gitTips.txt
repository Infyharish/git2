git add filename -- to add files to staged area

git commit -m "commit mesage"

git push filename 

git add .     --      to add all files instead of typing each filename separately.


git push origin {branch name}   --    mostly branch main .

git pull        --        to bring the changes that are made in git to vs code .


git branch        --      to check the current branch name we are working with .

git branch  -M main        --       used to change branch name according to our wish.


git branch -a         --       used to list all the branches that are present in the repo.


git checkout {branch name }         --       used to change from one branch to another.
